; Kin Programming Language grammar (BNF)
; Mirror of kin/grammar.bnf — derived from lexer + parser, not from older notes.
;
; Source of truth: kin/src/lexer/lexer.ts, kin/src/parser/parser.ts

program ::= statement*

statement ::= variableDeclaration
            | functionDeclaration
            | loopStatement
            | breakStatement
            | ifStatement
            | switchStatement
            | returnStatement
            | expressionStatement

; Semicolon only when the initializer is omitted.
; ntahinduka must have an initializer.
variableDeclaration ::= "reka" identifier (";" | "=" expression)
                      | "ntahinduka" identifier "=" expression

functionDeclaration ::= "porogaramu_ntoya" identifier "(" parameterList? ")" block

parameterList ::= identifier ("," identifier)*

block ::= "{" statement* "}"

loopStatement ::= "subiramo_niba" "(" expression ")" block

breakStatement ::= "hagarara" ";"?

ifStatement ::= "niba" "(" expression ")" block
                ("nanone_niba" "(" expression ")" block)*
                ("niba_byanze" block)?

; Determinant and case labels are primaryExpression only.
; A switch with only ibindi is accepted but its default body is discarded.
switchStatement ::= "gereranya" "(" primaryExpression ")" "{"
                      caseClause*
                      defaultClause?
                    "}"

caseClause    ::= "usanze" primaryExpression ":" statement*
defaultClause ::= "ibindi" ":" statement*

returnStatement ::= "tanga" ";"
                  | "tanga" expression

; No required semicolon.
expressionStatement ::= expression

; ---- expressions, low → high precedence ----

expression ::= assignmentExpression

assignmentExpression ::= objectLikeExpression ("=" assignmentExpression)?

objectLikeExpression ::= objectLiteral
                       | arrayLiteral
                       | logicalExpression

; Keys are identifiers only. Shorthand { key } is allowed. Trailing comma ok.
objectLiteral ::= "{" (objectProperty ("," objectProperty)* ","?)? "}"

objectProperty ::= identifier (":" expression)?

; Trailing comma ok. Arrays become objects keyed "0", "1", ...
arrayLiteral ::= "[" (expression ("," expression)* ","?)? "]"

; At most one && or ||. They do not chain.
logicalExpression ::= relationalExpression (("&&" | "||") relationalExpression)?

; At most one relational operator. They do not chain.
relationalExpression ::= additiveExpression (relationalOperator additiveExpression)?

relationalOperator ::= "<" | ">" | "==" | "!=" | "<=" | ">="

additiveExpression ::= multiplicativeExpression (additiveOperator multiplicativeExpression)*

additiveOperator ::= "+" | "-"

; * / % ^ share one left-associative precedence.
multiplicativeExpression ::= callMemberExpression (multiplicativeOperator callMemberExpression)*

multiplicativeOperator ::= "*" | "/" | "%" | "^"

; member then optional calls: obj.fn()()
; A call cannot be followed by further member access (foo().bar is invalid).
callMemberExpression ::= memberExpression ("(" argumentList? ")")*

argumentList ::= assignmentExpression ("," assignmentExpression)*

memberExpression ::= primaryExpression memberSuffix*

memberSuffix ::= "." identifier
               | "[" expression "]"

primaryExpression ::= identifier
                    | numericLiteral
                    | stringLiteral
                    | "(" expression ")"
                    | unaryNot

; ! applies only to an identifier.
unaryNot ::= "!" identifier

; ---- lexical ----

comment ::= "#" { anyCharacterExceptNewline }

identifier ::= (letter | "_") (letter | digit | "_")*

letter ::= "a"..."z" | "A"..."Z"
digit  ::= "0"..."9"

; Leading '-' only when immediately followed by a digit (lexer). -x is invalid.
numericLiteral ::= "-"? digit+ ("." digit+)?

; Double quotes only. No escapes. Newline in a string is an error.
stringLiteral ::= '"' { anyCharacterExceptQuoteOrNewline } '"'

; Keywords:
;   reka ntahinduka porogaramu_ntoya tanga
;   subiramo_niba hagarara
;   niba nanone_niba niba_byanze
;   gereranya usanze ibindi

; Lexer-only, not parsed:  ++  --  &  '     |  is a lexer error unless part of ||
